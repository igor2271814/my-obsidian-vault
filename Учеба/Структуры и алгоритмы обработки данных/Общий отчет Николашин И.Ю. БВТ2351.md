function fibonacciSearch(arr, x) {
  let fibMMm2 = 0; // (m-2)-ое число Фибоначчи
  let fibMMm1 = 1; // (m-1)-ое число Фибоначчи
  let fibM = fibMMm2 + fibMMm1; // m-ое число Фибоначчи

  while (fibM < arr.length) {
    fibMMm2 = fibMMm1;
    fibMMm1 = fibM;
    fibM = fibMMm2 + fibMMm1;
  }

  let offset = -1;

  while (fibM > 1) {
    const i = Math.min(offset + fibMMm2, arr.length - 1);

    if (arr[i] < x) {
      fibM = fibMMm1;
      fibMMm1 = fibMMm2;
      fibMMm2 = fibM - fibMMm1;
      offset = i;
    } else if (arr[i] > x) {
      fibM = fibMMm2;
      fibMMm1 = fibMMm1 - fibMMm2;
      fibMMm2 = fibM - fibMMm1;
    } else {
      return i;
    }
  }

  if (fibMMm1 && arr[offset + 1] === x) {
    return offset + 1;
  }

  return -1;
}

class BoyerMooreSearch {
  constructor(caseSensitive = true) {
    this.caseSensitive = caseSensitive;
  }

  // Формирование таблицы смещений
  buildBadCharTable(pattern) {
    const table = {};
    const length = pattern.length;
    for (let i = 0; i < length - 1; i++) {
      table[pattern[i]] = length - i - 1;
    }
    return table;
  }

  // Поиск подстроки
  search(text, pattern) {
    if (!this.caseSensitive) {
      text = text.toLowerCase();
      pattern = pattern.toLowerCase();
    }

    const badCharTable = this.buildBadCharTable(pattern);
    const patternLen = pattern.length;
    const textLen = text.length;
    let comparisons = 0;

    for (let i = 0; i <= textLen - patternLen;) {
      let j = patternLen - 1;
      while (j >= 0 && pattern[j] === text[i + j]) {
        j--;
        comparisons++;
      }

      if (j < 0) {
        return { position: i, comparisons };
      }

      const shift = badCharTable[text[i + patternLen - 1]] || patternLen;
      i += shift;
      comparisons++;
    }

    return { position: -1, comparisons };
  }
}

Ход работы
В соответствии с заданием, реализовали алгоритм на языке JavaScript
<!DOCTYPE html>
<html>

<head>
    <title>Кривая Гильберта</title>
    <style>
        body {
            font-family: Arial, sans-serif;
            text-align: center;
        }

        canvas {
            border: 1px solid #000;
            margin: 20px auto;
        }

        .controls {
            margin: 20px;
        }
    </style>
</head>

<body>
    <h1>Кривая Гильберта</h1>
    <div class="controls">
        <label>Глубина рекурсии: </label>
        <input type="number" id="depth" min="1" max="8" value="4">
        <button onclick="drawHilbert()">Построить</button>
        <div id="stats"></div>
    </div>
    <canvas id="hilbertCanvas" width="512" height="512"></canvas>

    <script>
        const canvas = document.getElementById('hilbertCanvas');
        const ctx = canvas.getContext('2d');
        let callCount = 0;
        let startTime = 0;

        function hilbert(x0, y0, xi, xj, yi, yj, n) {
            callCount++;
            if (n <= 0) {
                ctx.lineTo(x0 + (xi + yi) / 2, y0 + (xj + yj) / 2);
            } else {
                hilbert(x0, y0, yi / 2, yj / 2, xi / 2, xj / 2, n - 1);
                hilbert(x0 + xi / 2, y0 + xj / 2, xi / 2, xj / 2, yi / 2, yj / 2, n - 1);
                hilbert(x0 + xi / 2 + yi / 2, y0 + xj / 2 + yj / 2, xi / 2, xj / 2, yi / 2, yj / 2, n - 1);
                hilbert(x0 + xi / 2 + yi, y0 + xj / 2 + yj, -yi / 2, -yj / 2, -xi / 2, -xj / 2, n - 1);
            }
        }

        function drawHilbert() {
            const depth = parseInt(document.getElementById('depth').value);
            const size = Math.min(canvas.width, canvas.height) * 0.9;
            const x0 = (canvas.width - size) / 2;
            const y0 = (canvas.height - size) / 2;

            ctx.clearRect(0, 0, canvas.width, canvas.height);
            ctx.beginPath();
            ctx.strokeStyle = '#3498db';
            ctx.lineWidth = 2;

            callCount = 0;
            startTime = performance.now();

            hilbert(x0, y0, size, 0, 0, size, depth);

            const endTime = performance.now();
            const duration = (endTime - startTime).toFixed(2);

            ctx.stroke();

            document.getElementById('stats').innerHTML = `
                <p>Глубина: ${depth}</p>
                <p>Вызовов функции: ${callCount}</p>
                <p>Время выполнения: ${duration} мс</p>
            `;
        }

        // Инициализация при загрузке
        window.onload = drawHilbert;
    </script>
</body>

</html>
